SQL> -- 3. Each project tablespace with its size, autoextend and maximum size
SQL> COLUMN tablespace_name FORMAT A16
SQL> COLUMN contents	    FORMAT A10
SQL> COLUMN file_name	    FORMAT A90
SQL> SELECT t.tablespace_name, t.contents,
  2  	    f.bytes / 1024 / 1024			  AS size_mb,
  3  	    f.autoextensible,
  4  	    f.increment_by * t.block_size / 1024 / 1024   AS next_mb,
  5  	    f.maxbytes / 1024 / 1024			  AS max_mb,
  6  	    f.file_name
  7  FROM   dba_tablespaces t
  8  JOIN  (SELECT tablespace_name, file_name, bytes, autoextensible, increment_by, maxbytes FROM dba_data_files
  9  	    UNION ALL
 10  	    SELECT tablespace_name, file_name, bytes, autoextensible, increment_by, maxbytes FROM dba_temp_files) f
 11  	    ON f.tablespace_name = t.tablespace_name
 12  WHERE  t.tablespace_name LIKE 'PRICEWATCH%'
 13  ORDER  BY t.tablespace_name;
